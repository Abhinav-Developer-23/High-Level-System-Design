TPAP Internal Flow

flowchart TD A["POST /upi/ext/txn/lite/v2/app-pay Controller: UPILiteController"] --> B["LitePayService.requestPay()"] B --> DB1["DB[txn_info] INSERT status=PENDING"] B --> DB2["DB[txn_participants] INSERT Payer + Payee"] B --> C["async: GatewayAsyncHttpClientHelper"] C --> D["Juspay Gateway POST /lite/send-money"] D --> E["await callback POST /txn-callback"] E --> DB3["DB[txn_info] UPDATE status, rrn"]

flowchart TD A["POST /lite/send-money Controller: TransactionController"] --> B["Transform NPCIInTO to LiteRequest (Extract ARQC from credBlock)"] B --> C["LiteRequestAPIClient.execute()"] C --> D["Juspay API Call POST /sendMoney"] D --> E["Receive result from Juspay"] E --> F["TransactionCallbackClient POST /txn-callback to Switch"]
